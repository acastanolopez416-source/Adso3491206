/* Ejemplo de Propiedades Tipográficas */
h1 {
  color: #369;
  font-family: Arial, sans-serif;
  font-size: 2em;
  font-weight: bold;
  font-style: italic;
}

/* Propiedad Shorthand (font) */
p { font: italic bold 16px Arial, sans-serif; }
